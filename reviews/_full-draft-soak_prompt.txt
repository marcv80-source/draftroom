## The new file in full

"""A COMPLETE draft, driven through the real API against the real cached pool.

WHY THIS FILE EXISTS
--------------------
Bookkeeping is this tool's FIRST job. It is the only record of an in-person sticker-board draft,
so a projection being 8% off costs a marginal pick, while the clock drifting one slot at pick 47
costs the draft. Before this file, `tests/test_server.py` had 69 tests and **not one of them
drafted more than a handful of picks**: every mechanic was pinned in isolation and the thing that
actually happens on draft night -- 150 consecutive mutations against one event log -- had never
run in the suite at all.

That gap is not hypothetical. Two bugs of exactly this shape reached the repo and were caught by
review rather than by tests, both recorded in CLAUDE.md:

* ``out_of_order`` was computed as ``team_slot is not None``, so click-anywhere drafting (which
  always supplies a slot) badged EVERY ordinary pick as out of order.
* "Undraft" on the newest pick appended a ``pick_voided`` and left the clock advanced, so the
  replacement landed at the next pick number for the next team and the whole board drifted one
  slot against the physical one -- silently.

Neither is visible in a 3-pick test. Both are obvious in a 150-pick one.

WHAT IS ASSERTED, AND WHY EACH ONE
----------------------------------
After EVERY pick, not just at the end:

1. **The clock matches snake arithmetic.** ``slot_on_clock(teams, pick)`` is the physical room's
   own rule; a drift here is the board disagreeing with the people at the table.
2. **No player is drafted twice.** The failure mode that a missing pick creates -- a drafted
   player still looking available.
3. **Filled picks == picks made, and there are no gaps below the clock.** A gap means a pick was
   recorded at the wrong number.
4. **The drafted player is marked drafted on the board and owned by the right team.**

And at the end, the one that subsumes the rest: **replaying the event log from disk reproduces
the live state exactly**. That is literally what crash recovery does, so if it ever disagrees,
a relaunch mid-draft serves a different board than the one Marc was looking at.

Uses the REAL 10-team config and the REAL cached pool rather than a fixture, because the point is
end-to-end confidence in the thing that will run on draft night. Skips (never fails) when the
cache is absent, per this repo's convention for cache-dependent tests.
"""

from __future__ import annotations

import json

import pytest
from fastapi.testclient import TestClient

from draftroom import live_data
from draftroom.config import LeagueConfig
from draftroom.draft import snake
from draftroom.draft.events import EventLog
from draftroom.draft.state import DraftState
from draftroom.server import create_app

MY_SLOT = 4


@pytest.fixture(scope="module")
def real_cfg() -> LeagueConfig:
    return LeagueConfig.from_yaml()


@pytest.fixture(scope="module")
def real_pool():
    """The real 980-player pool, built ONCE -- it is the expensive part of this file."""
    try:
        pool = live_data.load_player_pool()
    except FileNotFoundError as exc:
        pytest.skip(f"no cached prep data for a real-pool soak: {exc}")
    if not pool:
        pytest.skip("cached pool is empty")
    return pool


def _client(tmp_path, real_cfg, real_pool, *, name: str = "draft.jsonl") -> TestClient:
    app = create_app(
        cfg=real_cfg, my_slot=MY_SLOT, log_path=tmp_path / name, pool=list(real_pool)
    )
    return TestClient(app)


def _rows(payload) -> list[dict]:
    return [r for pos in payload["tier_board"] for r in payload["tier_board"][pos]]


def _best_available(payload) -> dict:
    """The best undrafted RANKED player, i.e. what a manager plausibly takes.

    Deliberately a real choice rather than a random one: it walks the board down in value order,
    which keeps the draft realistic (early rounds drain the top, late rounds reach the tail) and
    makes a failure reproducible.
    """
    avail = [
        r for r in _rows(payload) if not r["drafted"] and r["is_ranked"] and r["value_is_real"]
    ]
    if not avail:
        avail = [r for r in _rows(payload) if not r["drafted"] and r["is_ranked"]]
    assert avail, "the pool ran dry before the draft finished"
    return max(avail, key=lambda r: r["value"])


def _assert_consistent(payload, *, picks_made: int, cfg: LeagueConfig) -> None:
    """Every invariant that must hold after any mutation. Called ~150 times."""
    total = cfg.teams * cfg.roster_size

    # 1 - the clock agrees with the room's own snake rule
    expected_clock = min(picks_made + 1, total + 1)
    assert payload["current_pick"] == expected_clock, (
        f"after {picks_made} picks the clock should be at {expected_clock}, "
        f"got {payload['current_pick']}"
    )
    if payload["current_pick"] <= total:
        assert payload["slot_on_clock"] == snake.slot_on_clock(
            cfg.teams, payload["current_pick"]
        ), "slot_on_clock drifted from snake arithmetic -- the board now disagrees with the table"

    live = [p for p in payload["all_picks"] if not p["voided"]]

    # 2 - nobody is drafted twice
    ids = [p["player_id"] for p in live if p["player_id"]]
    assert len(ids) == len(set(ids)), (
        f"duplicate player(s) on the board: "
        f"{sorted({i for i in ids if ids.count(i) > 1})}"
    )

    # 3 - the count reconciles and nothing was recorded at the wrong number
    assert len(live) == picks_made, f"expected {picks_made} live picks, found {len(live)}"
    assert payload["gaps"] == [], f"gaps below the clock: {payload['gaps']}"
    assert sorted(p["pick_no"] for p in live) == list(range(1, picks_made + 1)), (
        "live pick numbers are not the contiguous run 1..N"
    )


# --------------------------------------------------------------------------- the soak


def test_a_complete_draft_stays_consistent_at_every_single_pick(tmp_path, real_cfg, real_pool):
    """150 picks, checked after each one. THE test this file exists for."""
    client = _client(tmp_path, real_cfg, real_pool)
    total = real_cfg.teams * real_cfg.roster_size

    payload = client.get("/api/state").json()
    _assert_consistent(payload, picks_made=0, cfg=real_cfg)

    taken: list[tuple[int, str, int]] = []
    for pick_no in range(1, total + 1):
        on_clock = payload["slot_on_clock"]
        target = _best_available(payload)
        resp = client.post(
            "/api/pick",
            json={
                "player_id": target["player_id"],
                "pick_no": pick_no,
                "team_slot": on_clock,
            },
        )
        assert resp.status_code == 200, f"pick {pick_no} failed: {resp.text}"
        payload = resp.json()
        taken.append((pick_no, target["player_id"], on_clock))
        _assert_consistent(payload, picks_made=pick_no, cfg=real_cfg)

        # 4 - the player is now marked drafted, and owned by the team that took him
        row = next(r for r in _rows(payload) if r["player_id"] == target["player_id"])
        assert row["drafted"], f"{target['name']} was drafted but the board still shows him free"
        assert row["owner_team_slot"] == on_clock

    assert len(taken) == total

    # Every team ends with a full roster -- the arithmetic the room will actually eyeball.
    by_slot: dict[int, int] = {}
    for _, _, slot in taken:
        by_slot[slot] = by_slot.get(slot, 0) + 1
    assert by_slot == {s: real_cfg.roster_size for s in range(1, real_cfg.teams + 1)}, (
        f"uneven rosters at the end of a snake draft: {sorted(by_slot.items())}"
    )

    # THE strongest check: a cold replay of the log equals the live board. This is exactly what
    # a mid-draft relaunch does, so a disagreement here means crash recovery serves a different
    # draft than the one on screen.
    replayed = DraftState.replay(
        EventLog(tmp_path / "draft.jsonl").events(),
        teams=real_cfg.teams,
        rounds=real_cfg.roster_size,
        my_slot=MY_SLOT,
    )
    assert replayed.current_pick == payload["current_pick"]
    assert len(replayed.picks) == total
    for pick_no, player_id, slot in taken:
        got = replayed.picks[pick_no]
        assert got.player_id == player_id, f"replay disagrees on pick {pick_no}"
        assert got.team_slot == slot, f"replay disagrees on who made pick {pick_no}"


def test_the_whole_draft_is_one_appended_event_per_pick(tmp_path, real_cfg, real_pool):
    """The log is append-only and must not accumulate hidden extra events.

    A pick that writes two events can be torn in half by a crash -- the reason `undraft` appends
    ONE event rather than a void+clock_set pair (CLAUDE.md). Cheap to assert, and it would catch
    a well-meaning refactor that "helpfully" also stamps the clock.
    """
    client = _client(tmp_path, real_cfg, real_pool)
    payload = client.get("/api/state").json()
    n = 25
    for pick_no in range(1, n + 1):
        target = _best_available(payload)
        payload = client.post(
            "/api/pick",
            json={
                "player_id": target["player_id"],
                "pick_no": pick_no,
                "team_slot": payload["slot_on_clock"],
            },
        ).json()

    lines = [
        json.loads(line)
        for line in (tmp_path / "draft.jsonl").read_text(encoding="utf-8").splitlines()
        if line.strip()
    ]
    assert len(lines) == n, f"{n} picks wrote {len(lines)} events"
    assert all(ev["type"] == "pick" for ev in lines)


# --------------------------------------------------------------------------- messy reality


def _draft_n(client, payload, n: int, start: int = 1):
    """Draft n picks in order, returning (payload, [(pick_no, player_id, slot), ...])."""
    taken = []
    for pick_no in range(start, start + n):
        target = _best_available(payload)
        slot = payload["slot_on_clock"]
        payload = client.post(
            "/api/pick",
            json={"player_id": target["player_id"], "pick_no": pick_no, "team_slot": slot},
        ).json()
        taken.append((pick_no, target["player_id"], slot))
    return payload, taken


def test_undrafting_the_newest_pick_rewinds_the_clock_and_leaves_no_gap(
    tmp_path, real_cfg, real_pool
):
    """The board-drift bug, pinned mid-draft rather than at pick 1.

    Voiding the newest pick instead of undrafting it left the clock advanced, so the replacement
    landed at the next pick number for the NEXT team and every later pick sat one slot off the
    physical board. Nothing on screen said so.
    """
    client = _client(tmp_path, real_cfg, real_pool)
    payload = client.get("/api/state").json()
    payload, taken = _draft_n(client, payload, 20)
    last_pick, last_player, last_slot = taken[-1]

    resp = client.post("/api/undraft", json={"pick_no": last_pick})
    assert resp.status_code == 200, resp.text
    payload = resp.json()

    assert payload["last_undraft"]["mode"] == "undone", (
        "the newest pick must be UNDONE (clock rewinds), never voided (clock stays advanced)"
    )
    _assert_consistent(payload, picks_made=last_pick - 1, cfg=real_cfg)
    assert payload["slot_on_clock"] == last_slot, (
        "the clock must return to the team whose pick was removed"
    )
    row = next(r for r in _rows(payload) if r["player_id"] == last_player)
    assert not row["drafted"], "the undrafted player must be available again"

    # And the replacement lands at the SAME pick number for the SAME team.
    replacement = _best_available(payload)
    payload = client.post(
        "/api/pick",
        json={
            "player_id": replacement["player_id"],
            "pick_no": last_pick,
            "team_slot": payload["slot_on_clock"],
        },
    ).json()
    _assert_consistent(payload, picks_made=last_pick, cfg=real_cfg)
    landed = next(p for p in payload["all_picks"] if p["pick_no"] == last_pick and not p["voided"])
    assert landed["team_slot"] == last_slot


def test_undrafting_an_OLDER_pick_voids_it_and_reports_the_hole(tmp_path, real_cfg, real_pool):
    """The other half of the two-acts rule: an older pick cannot rewind the clock.

    Rewinding would renumber everything after it. So it is voided and the hole is REPORTED,
    because a silently missing pick makes a drafted player look available.
    """
    client = _client(tmp_path, real_cfg, real_pool)
    payload = client.get("/api/state").json()
    payload, taken = _draft_n(client, payload, 20)
    old_pick, old_player, _ = taken[5]
    clock_before = payload["current_pick"]

    resp = client.post("/api/undraft", json={"pick_no": old_pick})
    assert resp.status_code == 200, resp.text
    payload = resp.json()

    assert payload["last_undraft"]["mode"] == "voided", "an older pick must not rewind the clock"
    assert payload["current_pick"] == clock_before, "the clock moved on an older-pick undraft"
    assert old_pick in payload["gaps"], (
        f"pick {old_pick} was removed and must be reported as a gap, not vanish"
    )
    row = next(r for r in _rows(payload) if r["player_id"] == old_player)
    assert not row["drafted"], "the voided pick's player must be available again"


def test_an_out_of_order_pick_is_flagged_and_an_ordinary_one_is_not(
    tmp_path, real_cfg, real_pool
):
    """`out_of_order` must mean "this pick did not go to the team on the clock" -- nothing else.

    It was once computed as `team_slot is not None`, and since click-anywhere drafting always
    supplies a slot, that badged every ordinary pick. A 150-pick run makes the noise obvious; a
    3-pick one does not.
    """
    client = _client(tmp_path, real_cfg, real_pool)
    payload = client.get("/api/state").json()
    payload, _ = _draft_n(client, payload, 10)

    ordinary = [p for p in payload["all_picks"] if not p["voided"]]
    assert not any(p["out_of_order"] for p in ordinary), (
        "ten picks all made by the team on the clock; none may be flagged out of order"
    )

    on_clock = payload["slot_on_clock"]
    wrong_slot = (on_clock % real_cfg.teams) + 1
    target = _best_available(payload)
    payload = client.post(
        "/api/pick",
        json={"player_id": target["player_id"], "pick_no": 11, "team_slot": wrong_slot},
    ).json()

    flagged = next(p for p in payload["all_picks"] if p["pick_no"] == 11)
    assert flagged["out_of_order"], (
        f"pick 11 went to slot {wrong_slot} while slot {on_clock} was on the clock and must be "
        "flagged"
    )
    assert sum(1 for p in payload["all_picks"] if p["out_of_order"]) == 1


def test_reassigning_a_pick_moves_the_owner_and_not_the_player(tmp_path, real_cfg, real_pool):
    """Right-click -> "Reassign to team" -- the in-room fix for "that was actually Jay's pick"."""
    client = _client(tmp_path, real_cfg, real_pool)
    payload = client.get("/api/state").json()
    payload, taken = _draft_n(client, payload, 12)
    pick_no, player_id, slot = taken[4]
    new_slot = (slot % real_cfg.teams) + 1

    resp = client.post("/api/reassign", json={"pick_no": pick_no, "team_slot": new_slot})
    assert resp.status_code == 200, resp.text
    payload = resp.json()

    moved = next(p for p in payload["all_picks"] if p["pick_no"] == pick_no)
    assert moved["team_slot"] == new_slot
    assert moved["player_id"] == player_id, "a reassign must not touch the player"
    _assert_consistent(payload, picks_made=12, cfg=real_cfg)

def test_a_relaunch_mid_draft_serves_exactly_the_board_it_left(tmp_path, real_cfg, real_pool):
    """Closed lid, dead battery, or a stray Ctrl-C at pick 63.

    The event log is the only durable state, so a second app on the same log must come back with
    the identical draft. Anything less and a mid-draft restart quietly loses picks in a room where
    nobody can tell.
    """
    first = _client(tmp_path, real_cfg, real_pool)
    payload = first.get("/api/state").json()
    payload, taken = _draft_n(first, payload, 63)
    before = payload

    second = _client(tmp_path, real_cfg, real_pool)  # same log_path
    after = second.get("/api/state").json()

    assert after["current_pick"] == before["current_pick"] == 64
    assert after["slot_on_clock"] == before["slot_on_clock"]
    assert after["gaps"] == before["gaps"] == []

    def _key(p):
        return (p["pick_no"], p["player_id"], p["team_slot"], p["voided"])

    assert sorted(map(_key, after["all_picks"])) == sorted(map(_key, before["all_picks"]))
    _assert_consistent(after, picks_made=63, cfg=real_cfg)

    # And the resumed app can keep drafting, at the right pick, for the right team.
    resumed, _ = _draft_n(second, after, 5, start=64)
    _assert_consistent(resumed, picks_made=68, cfg=real_cfg)


def test_a_relaunch_after_an_undraft_does_not_resurrect_the_removed_pick(
    tmp_path, real_cfg, real_pool
):
    """The undo/void distinction has to survive replay, or crash recovery undoes the correction.

    This is the pairing that matters: `undraft` appends ONE event precisely so a crash cannot
    tear it, and replay is what proves the single event was enough.
    """
    first = _client(tmp_path, real_cfg, real_pool)
    payload = first.get("/api/state").json()
    payload, taken = _draft_n(first, payload, 30)
    last_pick, last_player, _ = taken[-1]
    payload = first.post("/api/undraft", json={"pick_no": last_pick}).json()

    second = _client(tmp_path, real_cfg, real_pool)
    after = second.get("/api/state").json()

    assert after["current_pick"] == last_pick, "the rewound clock must survive a relaunch"
    row = next(r for r in _rows(after) if r["player_id"] == last_player)
    assert not row["drafted"], "the undrafted player came back to life on replay"
    _assert_consistent(after, picks_made=last_pick - 1, cfg=real_cfg)

def test_the_late_rounds_can_record_UNRANKED_players(tmp_path, real_cfg, real_pool):
    """The pool is two tiers and the late rounds live in the second one.

    ~790 of the 980 players carry `is_ranked=False` and `value=0.0`, which is NOT an evaluation --
    they exist so the board can RECORD them (CLAUDE.md). Round 12 of a real draft is full of those
    names, so drafting one has to work and has to leave the bookkeeping intact.
    """
    client = _client(tmp_path, real_cfg, real_pool)
    payload = client.get("/api/state").json()
    payload, _ = _draft_n(client, payload, 10)

    unranked = [r for r in _rows(payload) if not r["is_ranked"] and not r["drafted"]]
    if not unranked:
        pytest.skip("this cached pool carries no unranked players")
    target = unranked[0]
    assert target["value"] == 0.0, "an unranked player carries no valuation, by design"

    resp = client.post(
        "/api/pick",
        json={
            "player_id": target["player_id"],
            "pick_no": 11,
            "team_slot": payload["slot_on_clock"],
        },
    )
    assert resp.status_code == 200, resp.text
    payload = resp.json()
    _assert_consistent(payload, picks_made=11, cfg=real_cfg)

    recorded = next(p for p in payload["all_picks"] if p["pick_no"] == 11)
    assert recorded["player_id"] == target["player_id"]
    assert recorded["name"] == target["name"], (
        "an unranked pick must record the real NAME -- bookkeeping is the point of tier two"
    )
    row = next(r for r in _rows(payload) if r["player_id"] == target["player_id"])
    assert row["drafted"], "an unranked player must be removed from the available board too"

def test_a_missed_pick_is_reported_as_a_gap_and_closes_when_filled(
    tmp_path, real_cfg, real_pool
):
    """Marc looks up, three picks have happened, and he only caught the last one.

    A gap is the honest state. It must be REPORTED, because the alternative -- a pick silently
    absent -- makes an already-drafted player look available, which is the single worst thing
    this board can do in a live room.
    """
    client = _client(tmp_path, real_cfg, real_pool)
    payload = client.get("/api/state").json()
    payload, _ = _draft_n(client, payload, 8)

    # Two picks happen at the table while he is talking to someone: jump the clock past them.
    payload = client.post("/api/clock", json={"pick_no": 11}).json()
    assert payload["gaps"] == [9, 10], f"missed picks 9 and 10 must be reported: {payload['gaps']}"

    # He records pick 11 normally. The holes stay visible -- they are not papered over.
    target = _best_available(payload)
    payload = client.post(
        "/api/pick",
        json={
            "player_id": target["player_id"],
            "pick_no": 11,
            "team_slot": payload["slot_on_clock"],
        },
    ).json()
    assert payload["gaps"] == [9, 10], "recording a later pick must not erase the earlier holes"

    # Then he backfills them from the physical board, and the gaps close.
    for pick_no in (9, 10):
        fill = _best_available(payload)
        payload = client.post(
            "/api/pick",
            json={
                "player_id": fill["player_id"],
                "pick_no": pick_no,
                "team_slot": snake.slot_on_clock(real_cfg.teams, pick_no),
            },
        ).json()
    assert payload["gaps"] == [], "backfilled picks must close the gaps"
    _assert_consistent(payload, picks_made=11, cfg=real_cfg)


# --------------------------------------------------------------------------- source toggle


def test_switching_projection_source_mid_draft_keeps_the_draft_and_survives_a_relaunch(
    tmp_path, real_cfg, real_pool
):
    """Marc changes his mind about the board at round 4 and the laptop dies at round 6.

    Two documented hazards meet here: `source_changed` is appended and fsync'd BEFORE the
    in-memory pool moves (so a failed write cannot leave the running app on one source while
    replay rebuilds another), and a relaunch must resume the source he CHOSE rather than
    silently reverting to the default (CLAUDE.md, draft-night hazards).

    Note this test injects a pool, and an injected pool always wins over the log on construction
    -- that is how the other tests stay hermetic. So what is asserted here is that the toggle
    itself is durable and the DRAFT survives it, which is the bookkeeping guarantee. The
    board-rebuild half of the resume is covered by tests/test_sources.py.
    """
    client = _client(tmp_path, real_cfg, real_pool)
    payload = client.get("/api/state").json()
    payload, taken = _draft_n(client, payload, 35)
    before_source = payload["active_source"]

    available = client.get("/api/sources").json()
    keys = [
        s["key"]
        for s in available.get("sources", [])
        if s.get("key") != before_source and s.get("available", True)
    ]
    if not keys:
        pytest.skip("only one projection source is available from this cache")

    resp = client.post("/api/source", json={"key": keys[0]})
    assert resp.status_code == 200, resp.text
    payload = resp.json()
    _assert_consistent(payload, picks_made=35, cfg=real_cfg)

    # The draft itself is untouched by a board change -- picks are facts, values are opinions.
    live = {p["pick_no"]: p["player_id"] for p in payload["all_picks"] if not p["voided"]}
    assert live == {n: pid for n, pid, _ in taken}

    # ...and drafting continues normally afterwards.
    payload, _ = _draft_n(client, payload, 5, start=36)
    _assert_consistent(payload, picks_made=40, cfg=real_cfg)

    # The choice is durable: it is in the log, ahead of nothing that contradicts it.
    events = [ev.type for ev in EventLog(tmp_path / "draft.jsonl").events()]
    assert "source_changed" in events, "the toggle must be recorded, or a relaunch loses it"

    relaunched = _client(tmp_path, real_cfg, real_pool).get("/api/state").json()
    assert relaunched["current_pick"] == 41
    assert relaunched["gaps"] == []
    _assert_consistent(relaunched, picks_made=40, cfg=real_cfg)


# --------------------------------------------------------------------------- interleaved


def test_a_full_draft_with_corrections_INTERLEAVED_stays_consistent(tmp_path, real_cfg, real_pool):
    """The closest thing in this suite to the actual room.

    Every other test here exercises one messy path on an otherwise clean draft. Draft night is
    not like that: a mis-entered pick gets undrafted and replaced two picks later, someone points
    out that a pick was actually Jay's, a name gets recorded against the wrong team because two
    people spoke at once. The bugs this file exists to catch were all ORDERING bugs -- a clock
    left advanced, an event replayed out of sequence -- and ordering bugs are exactly what only
    show up when the operations are interleaved.

    So: a complete 150-pick draft with a perturbation roughly every 17 picks, every invariant
    checked after every single mutation, and a cold replay compared to the live board at the end.
    The perturbation schedule is FIXED rather than random -- a soak test that fails once a week
    on a seed nobody recorded is worse than no test.
    """
    client = _client(tmp_path, real_cfg, real_pool)
    total = real_cfg.teams * real_cfg.roster_size
    payload = client.get("/api/state").json()

    # pick_no -> what to do after recording it. Chosen to land in different rounds, and to put
    # an undraft-and-replace immediately before a snake turn (where a clock error does the most
    # damage, because the same team picks twice in a row).
    UNDRAFT_AND_REPLACE = {19, 20, 61, 100, 139}
    REASSIGN = {33, 78, 121}
    OUT_OF_ORDER = {47, 94}

    expected: dict[int, tuple[str, int]] = {}  # pick_no -> (player_id, team_slot)
    perturbations = 0

    for pick_no in range(1, total + 1):
        on_clock = payload["slot_on_clock"]
        slot = on_clock
        if pick_no in OUT_OF_ORDER:
            # Recorded against the wrong team on purpose: it happens, and the board must take it
            # and FLAG it rather than refuse or silently accept.
            slot = (on_clock % real_cfg.teams) + 1
        target = _best_available(payload)
        resp = client.post(
            "/api/pick",
            json={"player_id": target["player_id"], "pick_no": pick_no, "team_slot": slot},
        )
        assert resp.status_code == 200, f"pick {pick_no} failed: {resp.text}"
        payload = resp.json()
        expected[pick_no] = (target["player_id"], slot)
        _assert_consistent(payload, picks_made=pick_no, cfg=real_cfg)

        if pick_no in OUT_OF_ORDER:
            flagged = next(p for p in payload["all_picks"] if p["pick_no"] == pick_no)
            assert flagged["out_of_order"], f"pick {pick_no} went to the wrong team unflagged"
            perturbations += 1

        if pick_no in UNDRAFT_AND_REPLACE:
            # "No wait, that was the wrong guy." The newest pick, so the clock must rewind and
            # the replacement must land at the SAME number for the SAME team.
            resp = client.post("/api/undraft", json={"pick_no": pick_no})
            assert resp.status_code == 200, resp.text
            payload = resp.json()
            assert payload["last_undraft"]["mode"] == "undone"
            _assert_consistent(payload, picks_made=pick_no - 1, cfg=real_cfg)
            assert payload["slot_on_clock"] == slot, (
                f"after undrafting pick {pick_no} the clock must return to slot {slot}"
            )
            freed = next(r for r in _rows(payload) if r["player_id"] == expected[pick_no][0])
            assert not freed["drafted"], "the undrafted player must be available again"

            replacement = _best_available(payload)
            payload = client.post(
                "/api/pick",
                json={
                    "player_id": replacement["player_id"],
                    "pick_no": pick_no,
                    "team_slot": slot,
                },
            ).json()
            expected[pick_no] = (replacement["player_id"], slot)
            _assert_consistent(payload, picks_made=pick_no, cfg=real_cfg)
            perturbations += 1

        if pick_no in REASSIGN:
            # "That was actually Jay's pick." Ownership moves; the player does not.
            old_player, old_slot = expected[pick_no]
            new_slot = (old_slot % real_cfg.teams) + 1
            resp = client.post(
                "/api/reassign", json={"pick_no": pick_no, "team_slot": new_slot}
            )
            assert resp.status_code == 200, resp.text
            payload = resp.json()
            expected[pick_no] = (old_player, new_slot)
            moved = next(p for p in payload["all_picks"] if p["pick_no"] == pick_no)
            assert moved["team_slot"] == new_slot
            assert moved["player_id"] == old_player, "a reassign must not touch the player"
            _assert_consistent(payload, picks_made=pick_no, cfg=real_cfg)
            perturbations += 1

    assert perturbations == len(UNDRAFT_AND_REPLACE) + len(REASSIGN) + len(OUT_OF_ORDER)

    # The live board matches what we believe we entered, pick by pick.
    live = {
        p["pick_no"]: (p["player_id"], p["team_slot"])
        for p in payload["all_picks"]
        if not p["voided"]
    }
    assert live == expected, "the live board disagrees with the sequence of entered picks"

    # And a cold replay of the log matches it too -- through 10 perturbations, including five
    # undo events that have to drop their own pick during replay.
    replayed = DraftState.replay(
        EventLog(tmp_path / "draft.jsonl").events(),
        teams=real_cfg.teams,
        rounds=real_cfg.roster_size,
        my_slot=MY_SLOT,
    )
    assert replayed.current_pick == payload["current_pick"]
    assert {
        n: (pk.player_id, pk.team_slot) for n, pk in replayed.picks.items() if not pk.voided
    } == expected, "crash recovery would serve a different draft than the one on screen"

    # Rosters still reconcile after three reassignments moved ownership around.
    by_slot: dict[int, int] = {}
    for _pid, slot in expected.values():
        by_slot[slot] = by_slot.get(slot, 0) + 1
    assert sum(by_slot.values()) == total
    assert set(by_slot) == set(range(1, real_cfg.teams + 1))
